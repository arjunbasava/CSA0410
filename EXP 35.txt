#include <stdio.h>

int main()
{
    int index[20];
    int n, i, block;

    printf("Enter number of file blocks: ");
    scanf("%d", &n);

    printf("Enter block numbers:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &index[i]);

    printf("\nIndex Block:\n");
    for(i = 0; i < n; i++)
        printf("Entry %d -> Block %d\n", i + 1, index[i]);

    printf("\nEnter block position to access: ");
    scanf("%d", &block);

    if(block >= 1 && block <= n)
        printf("File block %d is at disk block %d\n",
               block, index[block - 1]);
    else
        printf("Invalid block position.\n");

    return 0;
}
