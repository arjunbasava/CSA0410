#include <stdio.h>

int main()
{
    char user[2][20],file[2][2][20];
    int i,j;

    for(i=0;i<2;i++)
    {
        printf("Enter User Name: ");
        scanf("%s",user[i]);

        for(j=0;j<2;j++)
        {
            printf("Enter File %d: ",j+1);
            scanf("%s",file[i][j]);
        }
    }

    printf("\nDirectory Structure\n");

    for(i=0;i<2;i++)
    {
        printf("%s\n",user[i]);

        for(j=0;j<2;j++)
            printf("  %s\n",file[i][j]);
    }

    return 0;
}
