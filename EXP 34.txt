#include <stdio.h>

int main()
{
    int record[20];
    int n, i, search;

    printf("Enter number of records: ");
    scanf("%d", &n);

    printf("Enter record values:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &record[i]);

    printf("\nEnter record to access: ");
    scanf("%d", &search);

    printf("\nRecords accessed:\n");

    for(i = 0; i < n; i++)
    {
        printf("%d ", record[i]);

        if(record[i] == search)
        {
            printf("\nRecord %d found.", search);
            break;
        }
    }

    if(i == n)
        printf("\nRecord not found.");

    return 0;
}
