#include <stdio.h>

int main()
{
    char file[5][20];
    int i,n;

    printf("Enter number of files: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        printf("Enter file %d: ",i+1);
        scanf("%s",file[i]);
    }

    printf("\nFiles in Directory:\n");

    for(i=0;i<n;i++)
        printf("%s\n",file[i]);

    return 0;
}
